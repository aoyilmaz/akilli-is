class BaseWindow(QMainWindow):
    """Tüm modül pencerelerinin temel sınıfı"""
    
    def __init__(self, db_path: str, title: str = "Akıllı İş ERP"):
        super().__init__()
        self.db_path = db_path
        self.setWindowTitle(title)
        self.setMinimumSize(1200, 800)
        
        # Ana widget ve layout
        self.central_widget = QWidget()
        self.setCentralWidget(self.central_widget)
        self.main_layout = QVBoxLayout(self.central_widget)
        
        # Toolbar ve Statusbar
        self.setup_toolbar()
        self.setup_statusbar()
        
    def setup_toolbar(self):
        """Standart toolbar oluştur"""
        self.toolbar = QToolBar("Ana Araç Çubuğu")
        self.addToolBar(self.toolbar)
        
        # Standart aksiyonlar
        self.action_new = QAction("Yeni", self)
        self.action_save = QAction("Kaydet", self)
        self.action_delete = QAction("Sil", self)
        self.action_refresh = QAction("Yenile", self)
        self.action_export = QAction("Dışa Aktar", self)
        
        self.toolbar.addAction(self.action_new)
        self.toolbar.addAction(self.action_save)
        self.toolbar.addAction(self.action_delete)
        self.toolbar.addSeparator()
        self.toolbar.addAction(self.action_refresh)
        self.toolbar.addAction(self.action_export)
        
    def setup_statusbar(self):
        """Status bar oluştur"""
        self.statusbar = QStatusBar()
        self.setStatusBar(self.statusbar)
        self.statusbar.showMessage("Hazır")
        
    def get_db_connection(self):
        """Veritabanı bağlantısı al"""
        return sqlite3.connect(self.db_path)
    
    def show_error(self, message: str):
        """Hata mesajı göster"""
        QMessageBox.critical(self, "Hata", message)
        
    def show_info(self, message: str):
        """Bilgi mesajı göster"""
        QMessageBox.information(self, "Bilgi", message)
        
    def confirm_action(self, message: str) -> bool:
        """Onay iste"""
        reply = QMessageBox.question(self, "Onay", message,
                                     QMessageBox.StandardButton.Yes | 
                                     QMessageBox.StandardButton.No)
        return reply == QMessageBox.StandardButton.Yes

    def load_bom_list(self):
        """BOM listesini veritabanından yükle"""
        conn = self.get_db_connection()
        cursor = conn.cursor()
        cursor.execute("""
            SELECT bh.bom_id, m.material_code, m.description, bh.alternative_bom
            FROM bom_headers bh
            JOIN materials m ON bh.material_id = m.material_id
            WHERE bh.status = 'ACTIVE'
            ORDER BY m.material_code
        """)
        rows = cursor.fetchall()
        conn.close()
        
        self.bom_list.setRowCount(len(rows))
        for i, row in enumerate(rows):
            self.bom_list.setItem(i, 0, QTableWidgetItem(row[1]))
            self.bom_list.setItem(i, 1, QTableWidgetItem(row[2]))
            self.bom_list.setItem(i, 2, QTableWidgetItem(str(row[3])))
            self.bom_list.item(i, 0).setData(Qt.ItemDataRole.UserRole, row[0])
            
    def load_bom_tree(self, bom_id: int, parent_item=None):
        """BOM ağacını recursive olarak yükle"""
        conn = self.get_db_connection()
        cursor = conn.cursor()
        
        if parent_item is None:
            # Root level
            self.bom_tree.clear()
            cursor.execute("""
                SELECT bi.bom_item_id, m.material_code, m.description, bi.quantity,
                       u.uom_code, bi.scrap_pct, bi.operation_no, bi.item_category,
                       m.material_id
                FROM bom_items bi
                JOIN materials m ON bi.component_id = m.material_id
                LEFT JOIN uom u ON bi.uom_id = u.uom_id
                WHERE bi.bom_id = ?
                ORDER BY bi.item_no
            """, (bom_id,))
        else:
            # Child level - component'ın kendi BOM'u var mı?
            material_id = parent_item.data(0, Qt.ItemDataRole.UserRole + 1)
            cursor.execute("""
                SELECT bh.bom_id FROM bom_headers bh
                WHERE bh.material_id = ? AND bh.status = 'ACTIVE'
            """, (material_id,))
            child_bom = cursor.fetchone()
            
            if child_bom:
                cursor.execute("""
                    SELECT bi.bom_item_id, m.material_code, m.description, bi.quantity,
                           u.uom_code, bi.scrap_pct, bi.operation_no, bi.item_category,
                           m.material_id
                    FROM bom_items bi
                    JOIN materials m ON bi.component_id = m.material_id
                    LEFT JOIN uom u ON bi.uom_id = u.uom_id
                    WHERE bi.bom_id = ?
                    ORDER BY bi.item_no
                """, (child_bom[0],))
            else:
                conn.close()
                return
        
        rows = cursor.fetchall()
        conn.close()
        
        for row in rows:
            item = QTreeWidgetItem([
                row[1],  # Bileşen Kodu
                row[2],  # Bileşen Adı
                f"{row[3]:.4f}",  # Miktar
                row[4] or "",  # Birim
                f"{row[5]:.2f}%" if row[5] else "",  # Fire
                str(row[6]) if row[6] else "",  # Operasyon No
                row[7] or ""  # Kategori
            ])
            item.setData(0, Qt.ItemDataRole.UserRole, row[0])  # bom_item_id
            item.setData(0, Qt.ItemDataRole.UserRole + 1, row[8])  # material_id
            
            if parent_item is None:
                self.bom_tree.addTopLevelItem(item)
            else:
                parent_item.addChild(item)
            
            # Recursive: alt seviyeleri yükle
            self.load_bom_tree(bom_id, item)
            
    def add_component(self):
        """Yeni bileşen ekle dialog"""
        dialog = ComponentDialog(self.db_path, self)
        if dialog.exec() == QDialog.DialogCode.Accepted:
            component_data = dialog.get_data()
            self.save_component(component_data)
            self.load_bom_tree(self.current_bom_id)
            
    def show_where_used(self):
        """Seçili bileşenin nerede kullanıldığını göster"""
        current = self.bom_tree.currentItem()
        if not current:
            self.show_error("Lütfen bir bileşen seçin")
            return
        material_id = current.data(0, Qt.ItemDataRole.UserRole + 1)
        dialog = WhereUsedDialog(self.db_path, material_id, self)
        dialog.exec()

3.3 MRP Çalıştırma Ekranı

# screens/mrp_run.py
 
class MRPWorker(QThread):
    """MRP hesaplamasını arka planda çalıştıran thread"""
    progress = pyqtSignal(int, str)  # yüzde, mesaj
    finished = pyqtSignal(dict)  # sonuç
    error = pyqtSignal(str)
    
    def __init__(self, db_path: str, params: dict):
        super().__init__()
        self.db_path = db_path
        self.params = params
        
    def run(self):
        try:
            from core.mrp_engine import MRPEngine
            engine = MRPEngine(self.db_path, self.params)
            engine.progress_callback = lambda p, m: self.progress.emit(p, m)
            result = engine.run_mrp()
            self.finished.emit(result)
        except Exception as e:
            self.error.emit(str(e))

    def run_mrp(self):
        """MRP çalıştır"""
        if self.mrp_worker and self.mrp_worker.isRunning():
            return
            
        params = {
            'run_type': self.cmb_run_type.currentIndex(),
            'horizon_days': self.spin_horizon.value(),
            'plant_id': self.cmb_plant.currentData(),
            'material_filter': self.cmb_material_filter.currentIndex(),
            'simulation': self.chk_simulation.isChecked()
        }
        
        self.btn_run.setEnabled(False)
        self.btn_cancel.setEnabled(True)
        self.progress_bar.setValue(0)
        
        self.mrp_worker = MRPWorker(self.db_path, params)
        self.mrp_worker.progress.connect(self.on_mrp_progress)
        self.mrp_worker.finished.connect(self.on_mrp_finished)
        self.mrp_worker.error.connect(self.on_mrp_error)
        self.mrp_worker.start()
        
    def on_mrp_progress(self, percent: int, message: str):
        """MRP ilerleme güncellemesi"""
        self.progress_bar.setValue(percent)
        self.lbl_status.setText(message)
        
    def on_mrp_finished(self, result: dict):
        """MRP tamamlandı"""
        self.btn_run.setEnabled(True)
        self.btn_cancel.setEnabled(False)
        self.progress_bar.setValue(100)
        self.lbl_status.setText("Tamamlandı")
        
        # Sonuçları göster
        self.update_summary_cards(result)
        self.load_planned_orders(result.get('planned_orders', []))
        self.load_exceptions(result.get('exceptions', []))

        """Durum değişikliği"""
        current_idx = self.STATUS_FLOW.index(self.current_status)
        new_idx = self.STATUS_FLOW.index(new_status)
        
        if new_idx != current_idx + 1:
            self.show_error("Geçersiz durum geçişi")
            return
            
        if not self.confirm_action(f"Durum '{self.STATUS_LABELS[new_status]}' olarak değiştirilsin mi?"):
            return
            
        # Veritabanı güncelle
        conn = self.get_db_connection()
        cursor = conn.cursor()
        cursor.execute("""
            UPDATE production_orders 
            SET status = ?, updated_at = ?
            WHERE production_order_id = ?
        """, (new_status, datetime.now(), self.order_id))
        conn.commit()
        conn.close()
        
        self.current_status = new_status
        self.update_status_ui()

            self.load_order(self.order_id)  # Refresh

3.6 Gantt Chart Çizelgeleme Ekranı

# screens/gantt_chart.py
 
from PyQt6.QtWidgets import QGraphicsView, QGraphicsScene, QGraphicsRectItem, QGraphicsTextItem
from PyQt6.QtCore import QRectF, QDateTime
from PyQt6.QtGui import QBrush, QColor, QPen, QFont
from datetime import datetime, timedelta
 
 
class GanttChartView(QGraphicsView):
    """Özel Gantt Chart görünümü"""
    
    HOUR_WIDTH = 30  # Her saat için piksel genişliği
    ROW_HEIGHT = 40
    HEADER_HEIGHT = 60
    
    STATUS_COLORS = {
        'CREATED': QColor("#E0E0E0"),
        'RELEASED': QColor("#BBDEFB"),
        'STARTED': QColor("#FFF59D"),
        'CONFIRMED': QColor("#A5D6A7"),
        'TECO': QColor("#81C784"),
        'CLOSED': QColor("#4CAF50")
    }
    
    def __init__(self):
        super().__init__()
        self.scene = QGraphicsScene()
        self.setScene(self.scene)
        self.setDragMode(QGraphicsView.DragMode.ScrollHandDrag)
        
        self.start_date = datetime.now().replace(hour=0, minute=0, second=0)
        self.days_to_show = 14
        self.resources = []  # İş merkezleri veya üretim emirleri
        self.tasks = []  # Operasyonlar
        
    def set_data(self, resources: list, tasks: list):
        """Gantt verilerini ayarla"""
        self.resources = resources
        self.tasks = tasks
        self.draw_chart()
        
    def draw_chart(self):
        """Gantt grafiğini çiz"""
        self.scene.clear()
        
        total_hours = self.days_to_show * 24
        chart_width = total_hours * self.HOUR_WIDTH
        chart_height = self.HEADER_HEIGHT + len(self.resources) * self.ROW_HEIGHT
        
        # Arka plan
        self.scene.setSceneRect(0, 0, chart_width, chart_height)
        
        # Zaman başlıkları çiz
        self.draw_time_header(chart_width)
        
        # Grid çizgileri
        self.draw_grid(chart_width, chart_height)
        
        # Resource satırları
        self.draw_resource_rows()
        
        # Task çubukları
        self.draw_tasks()
        
        # Bugün çizgisi
        self.draw_today_line(chart_height)
        
    def draw_time_header(self, width):
        """Zaman başlığı çiz"""
        # Gün başlıkları
        for day in range(self.days_to_show):
            date = self.start_date + timedelta(days=day)
            x = day * 24 * self.HOUR_WIDTH
            
            # Gün kutusu
            rect = QGraphicsRectItem(x, 0, 24 * self.HOUR_WIDTH, 30)
            rect.setBrush(QBrush(QColor("#1E3A5F")))
            self.scene.addItem(rect)
            
            # Gün yazısı
            text = QGraphicsTextItem(date.strftime("%d %b %Y - %A"))
            text.setDefaultTextColor(QColor("white"))
            text.setFont(QFont("Arial", 10, QFont.Weight.Bold))
            text.setPos(x + 10, 5)
            self.scene.addItem(text)
            
        # Saat başlıkları
        for hour in range(self.days_to_show * 24):
            x = hour * self.HOUR_WIDTH
            h = hour % 24
            
            if h % 4 == 0:  # Her 4 saatte bir göster
                text = QGraphicsTextItem(f"{h:02d}:00")
                text.setFont(QFont("Arial", 8))
                text.setPos(x + 2, 35)
                self.scene.addItem(text)
                
    def draw_grid(self, width, height):
        """Grid çizgileri çiz"""
        pen = QPen(QColor("#E0E0E0"))
        pen.setWidth(1)
        
        # Yatay çizgiler (resource satırları)
        for i in range(len(self.resources) + 1):
            y = self.HEADER_HEIGHT + i * self.ROW_HEIGHT
            line = self.scene.addLine(0, y, width, y, pen)
            
        # Dikey çizgiler (günler)
        for day in range(self.days_to_show + 1):
            x = day * 24 * self.HOUR_WIDTH
            line = self.scene.addLine(x, 0, x, height, pen)
            
    def draw_resource_rows(self):
        """Resource (iş merkezi/üretim emri) satırlarını çiz"""
        for i, resource in enumerate(self.resources):
            y = self.HEADER_HEIGHT + i * self.ROW_HEIGHT
            
            # Arka plan (alternatif renkler)
            bg_color = QColor("#FAFAFA") if i % 2 == 0 else QColor("#FFFFFF")
            rect = QGraphicsRectItem(0, y, 150, self.ROW_HEIGHT)
            rect.setBrush(QBrush(bg_color))
            rect.setPen(QPen(Qt.PenStyle.NoPen))
            self.scene.addItem(rect)
            
            # Resource adı
            text = QGraphicsTextItem(resource['name'][:20])
            text.setFont(QFont("Arial", 9))
            text.setPos(5, y + 10)
            self.scene.addItem(text)
            
    def draw_tasks(self):
        """Task çubuklarını çiz"""
        for task in self.tasks:
            resource_idx = self.get_resource_index(task['resource_id'])
            if resource_idx < 0:
                continue
                
            # Zaman hesaplama
            start_hours = self.datetime_to_hours(task['start_date'])
            end_hours = self.datetime_to_hours(task['end_date'])
            
            if start_hours < 0 or end_hours > self.days_to_show * 24:
                continue  # Görünür aralık dışında
                
            x = start_hours * self.HOUR_WIDTH
            width = (end_hours - start_hours) * self.HOUR_WIDTH
            y = self.HEADER_HEIGHT + resource_idx * self.ROW_HEIGHT + 5
            height = self.ROW_HEIGHT - 10
            
            # Task çubuğu
            color = self.STATUS_COLORS.get(task['status'], QColor("#E0E0E0"))
            rect = GanttTaskItem(x, y, width, height, task, self)
            rect.setBrush(QBrush(color))
            rect.setPen(QPen(QColor("#666666")))
            self.scene.addItem(rect)
            
            # Task etiketi
            if width > 50:
                text = QGraphicsTextItem(task['label'][:15])
                text.setFont(QFont("Arial", 8))
                text.setPos(x + 3, y + 3)
                self.scene.addItem(text)
                
    def draw_today_line(self, height):
        """Bugün çizgisi çiz"""
        now = datetime.now()
        hours = self.datetime_to_hours(now)
        
        if 0 <= hours <= self.days_to_show * 24:
            x = hours * self.HOUR_WIDTH
            pen = QPen(QColor("#D32F2F"))
            pen.setWidth(2)
            pen.setStyle(Qt.PenStyle.DashLine)
            self.scene.addLine(x, 0, x, height, pen)
            
    def datetime_to_hours(self, dt):
        """Datetime'ı başlangıçtan itibaren saat cinsine çevir"""
        delta = dt - self.start_date
        return delta.total_seconds() / 3600
 
 
class GanttTaskItem(QGraphicsRectItem):
    """Drag edilebilir Gantt task çubuğu"""
    
    def __init__(self, x, y, width, height, task_data, view):
        super().__init__(x, y, width, height)
        self.task_data = task_data
        self.view = view
        self.setFlag(QGraphicsRectItem.GraphicsItemFlag.ItemIsMovable)
        self.setFlag(QGraphicsRectItem.GraphicsItemFlag.ItemIsSelectable)
        self.setAcceptHoverEvents(True)
        
    def mouseReleaseEvent(self, event):
        """Sürükle bırak sonrası yeni pozisyonu kaydet"""
        new_x = self.pos().x()
        new_hours = new_x / self.view.HOUR_WIDTH
        new_start = self.view.start_date + timedelta(hours=new_hours)
        
        # Callback: yeni tarihi kaydet
        # self.view.on_task_moved(self.task_data, new_start)
        super().mouseReleaseEvent(event)

    procurement_type VARCHAR(10),  -- 'MAKE', 'BUY', 'BOTH'
    planning_strategy VARCHAR(10),  -- 'MTS', 'MTO', 'ATO'
    lot_size_type VARCHAR(15),  -- 'LOT_FOR_LOT', 'FIXED', 'EOQ'
    lot_size_qty DECIMAL(15,4),
    safety_stock DECIMAL(15,4) DEFAULT 0,
    lead_time_days INTEGER DEFAULT 0,
    standard_cost DECIMAL(15,4),

    updated_at TIMESTAMP
);

);
